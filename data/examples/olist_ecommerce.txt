Q: What is the total revenue generated in 2018?
SQL: SELECT ROUND(SUM(oi.price), 2) AS total_revenue
     FROM order_items oi
     JOIN orders o ON o.order_id = oi.order_id
     WHERE STRFTIME('%Y', o.order_purchase_timestamp) = '2018'
     LIMIT 200;

Q: Which product category has the highest average order value?
SQL: SELECT p.product_category_name,
            ROUND(AVG(oi.price), 2) AS avg_order_value
     FROM order_items oi
     JOIN products p ON p.product_id = oi.product_id
     WHERE p.product_category_name IS NOT NULL
     GROUP BY p.product_category_name
     ORDER BY avg_order_value DESC
     LIMIT 1;

Q: What is the average delivery time in days across all orders?
SQL: SELECT ROUND(AVG(JULIANDAY(order_delivered_customer_date) - JULIANDAY(order_purchase_timestamp)), 2) AS avg_delivery_days
     FROM orders
     WHERE order_delivered_customer_date IS NOT NULL
       AND order_purchase_timestamp IS NOT NULL
     LIMIT 200;

Q: Show the top 10 sellers by total revenue
SQL: SELECT oi.seller_id,
            ROUND(SUM(oi.price), 2) AS total_revenue
     FROM order_items oi
     GROUP BY oi.seller_id
     ORDER BY total_revenue DESC
     LIMIT 10;

Q: What percentage of orders have a 5-star review score?
SQL: SELECT ROUND(
         100.0 * SUM(CASE WHEN review_score = 5 THEN 1 ELSE 0 END) / COUNT(*), 2
     ) AS pct_five_star
     FROM order_reviews
     LIMIT 200;

Q: Which payment method is most commonly used?
SQL: SELECT payment_type, COUNT(*) AS payment_count
     FROM order_payments
     GROUP BY payment_type
     ORDER BY payment_count DESC
     LIMIT 1;

Q: Which state has the most customers?
SQL: SELECT customer_state, COUNT(*) AS customer_count
     FROM customers
     GROUP BY customer_state
     ORDER BY customer_count DESC
     LIMIT 1;

Q: Show monthly order counts for 2018
SQL: SELECT STRFTIME('%Y-%m', order_purchase_timestamp) AS year_month,
            COUNT(*) AS order_count
     FROM orders
     WHERE STRFTIME('%Y', order_purchase_timestamp) = '2018'
     GROUP BY year_month
     ORDER BY year_month ASC
     LIMIT 200;

Q: How many orders were delivered late (after estimated delivery date)?
SQL: SELECT COUNT(*) AS late_orders
     FROM orders
     WHERE order_delivered_customer_date > order_estimated_delivery_date
       AND order_delivered_customer_date IS NOT NULL
     LIMIT 200;

Q: What is the average number of items per order?
SQL: SELECT ROUND(CAST(COUNT(*) AS REAL) / COUNT(DISTINCT order_id), 2) AS avg_items_per_order
     FROM order_items
     LIMIT 200;

Q: What percentage of orders were cancelled?
SQL: SELECT ROUND(

     ) AS pct_cancelled
     FROM orders
     LIMIT 200;

Q: What is the average review score by product category?
SQL: SELECT p.product_category_name,
            ROUND(AVG(r.review_score), 2) AS avg_review_score,
            COUNT(r.review_id) AS review_count
     FROM order_reviews r
     JOIN orders o ON o.order_id = r.order_id
     JOIN order_items oi ON oi.order_id = o.order_id
     JOIN products p ON p.product_id = oi.product_id
     WHERE p.product_category_name IS NOT NULL
     GROUP BY p.product_category_name
     ORDER BY avg_review_score DESC
     LIMIT 200;

Q: What is the average freight value by seller state?
SQL: SELECT s.seller_state,
            ROUND(AVG(oi.freight_value), 2) AS avg_freight
     FROM order_items oi
     JOIN sellers s ON s.seller_id = oi.seller_id
     GROUP BY s.seller_state
     ORDER BY avg_freight DESC
     LIMIT 200;

Q: Which sellers have an average review score below 3?
SQL: SELECT oi.seller_id,
            ROUND(AVG(r.review_score), 2) AS avg_score
     FROM order_items oi
     JOIN orders o ON o.order_id = oi.order_id
     JOIN order_reviews r ON r.order_id = o.order_id
     GROUP BY oi.seller_id
     HAVING AVG(r.review_score) < 3
     ORDER BY avg_score ASC
     LIMIT 200;

Q: Show the top 5 states by average order value
SQL: SELECT c.customer_state,
            ROUND(AVG(oi.price), 2) AS avg_order_value
     FROM orders o
     JOIN customers c ON c.customer_id = o.customer_id
     JOIN order_items oi ON oi.order_id = o.order_id
     GROUP BY c.customer_state
     ORDER BY avg_order_value DESC
     LIMIT 5;

Q: Which product category has the most orders?
SQL: SELECT p.product_category_name,
            COUNT(DISTINCT oi.order_id) AS order_count
     FROM order_items oi
     JOIN products p ON p.product_id = oi.product_id
     WHERE p.product_category_name IS NOT NULL
     GROUP BY p.product_category_name
     ORDER BY order_count DESC
     LIMIT 1;
